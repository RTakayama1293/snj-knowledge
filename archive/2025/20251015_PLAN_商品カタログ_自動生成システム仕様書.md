## 1. プロジェクト概要

### 1.1 背景
- **現状**: 59社500商品を管理、月間最大100商品が増加
- **課題**: 
  - 見積書形式の商品カタログが見づらい（表形式で情報過密）
  - 商品追加のたびに手作業でカタログ更新
  - EEZOへの商品登録も別途手作業
- **目指す姿**: 商品マスタExcelを編集→ボタン一つでカタログ自動生成

### 1.2 プロジェクトゴール
1. **商品マスタの一元管理**: Excel 1ファイルで全商品情報を管理
2. **カタログ自動生成**: PowerPoint形式のカタログを自動生成・PDF化
3. **条件付き発行**: 必要な商品だけ選択してカタログ発行可能
4. **自走可能**: 非エンジニア（門永さん）がダブルクリックだけで実行可能

### 1.3 スコープ
- **Phase 1（今回）**: 商品マスタ設計 + カタログ自動生成
- **Phase 2（将来）**: EEZO連携、見積システム連携、価格表自動生成

### 7.2 excel_reader.py - Excelデータ読み込み

```python
"""
Excelファイルから商品データを読み込むモジュール
"""
import openpyxl
from pathlib import Path
from typing import List, Dict
import config

class ExcelReader:
    """商品マスタExcelを読み込むクラス"""
    
    def __init__(self, excel_path: Path = config.EXCEL_PATH):
        self.excel_path = excel_path
        self.workbook = None
        
    def load(self):
        """Excelファイルを読み込む"""
        if not self.excel_path.exists():
            raise FileNotFoundError(f"Excelファイルが見つかりません: {self.excel_path}")
        
        self.workbook = openpyxl.load_workbook(self.excel_path, data_only=True)
        print(f"✓ Excelファイル読み込み完了: {self.excel_path}")
    
    def get_products(self) -> List[Dict]:
        """商品情報を取得"""
        if not self.workbook:
            self.load()
        
        sheet = self.workbook[config.SHEET_PRODUCTS]
        
        # ヘッダー行を取得
        headers = [cell.value for cell in sheet[1]]
        
        # データ行を取得
        products = []
        for row in sheet.iter_rows(min_row=2, values_only=True):
            if row[0] is None:  # 商品IDが空の行はスキップ
                continue
            
            product = dict(zip(headers, row))
            
            # カタログ掲載フラグがTrueの商品のみ
            if product.get('カタログ掲載') in [True, 'TRUE', '○', 1]:
                products.append(product)
        
        print(f"✓ 商品データ取得完了: {len(products)}件")
        return products
    
    def get_makers(self) -> List[Dict]:
        """メーカー情報を取得"""
        if not self.workbook:
            self.load()
        
        sheet = self.workbook[config.SHEET_MAKERS]
        headers = [cell.value for cell in sheet[1]]
        
        makers = []
        for row in sheet.iter_rows(min_row=2, values_only=True):
            if row[0] is None:
                continue
            maker = dict(zip(headers, row))
            makers.append(maker)
        
        print(f"✓ メーカー情報取得完了: {len(makers)}件")
        return makers
    
    def get_config(self) -> Dict:
        """カタログ設定を取得"""
        if not self.workbook:
            self.load()
        
        sheet = self.workbook[config.SHEET_CONFIG]
        
        config_dict = {}
        for row in sheet.iter_rows(min_row=2, values_only=True):
            if row[0] is None:
                continue
            config_dict[row[0]] = row[1]
        
        print(f"✓ カタログ設定取得完了")
        return config_dict
    
    def validate_products(self, products: List[Dict]) -> tuple:
        """商品データをバリデーション"""
        valid_products = []
        invalid_products = []
        
        for product in products:
            missing_fields = []
            
            # 必須項目チェック
            for field in config.CATALOG_REQUIRED_FIELDS:
                if not product.get(field):
                    missing_fields.append(field)
            
            # 画像ファイル存在チェック
            if product.get('画像ファイル名'):
                image_path = config.IMAGES_DIR / product['画像ファイル名']
                if not image_path.exists():
                    missing_fields.append(f"画像ファイル({product['画像ファイル名']})")
            
            if missing_fields:
                invalid_products.append({
                    'product': product,
                    'missing_fields': missing_fields
                })
            else:
                valid_products.append(product)
        
        if invalid_products:
            print(f"⚠ バリデーション警告: {len(invalid_products)}件の商品に不備があります")
            for item in invalid_products[:5]:  # 最初の5件のみ表示
                print(f"  - {item['product'].get('Product_name', '不明')}: {', '.join(item['missing_fields'])}")
        
        print(f"✓ バリデーション完了: 有効{len(valid_products)}件、無効{len(invalid_products)}件")
        return valid_products, invalid_products
```

### 7.3 pptx_generator.py - PowerPoint生成

"""
PowerPointカタログを生成するモジュール
"""
from pptx import Presentation
from pptx.util import Inches, Pt
from pptx.enum.text import PP_ALIGN
from pptx.dml.color import RGBColor
from PIL import Image
from pathlib import Path
from datetime import datetime
from typing import List, Dict
import config

class CatalogGenerator:
    """カタログPowerPointを生成するクラス"""
    
    def __init__(self):
        self.prs = Presentation()
        self.prs.slide_width = Inches(config.SLIDE_WIDTH)

    def create_cover_page(self, catalog_config: Dict):

        title_para.font.bold = True
        title_para.font.name = config.FONT_NAME

    def create_maker_intro_page(self, maker: Dict):

        blank_layout = self.prs.slide_layouts[6]

        # メーカー名
        maker_name_box = slide.shapes.add_textbox(

        name_para.font.bold = True
        name_para.font.name = config.FONT_NAME

    def create_product_cards_page(self, products: List[Dict], page_num: int):
        """商品カードページを作成（2列×3行 = 6商品）"""
        blank_layout = self.prs.slide_layouts[6]
        slide = self.prs.slides.add_slide(blank_layout)
        
        # レイアウト設定
        card_width = Inches(4.5)
        card_height = Inches(2.3)
        margin_x = Inches(0.5)
        margin_y = Inches(0.5)
        gap_x = Inches(0.2)
        gap_y = Inches(0.2)
        
        for idx, product in enumerate(products[:6]):  # 最大6商品
            row = idx // 2
            col = idx % 2
            
            x = margin_x + col * (card_width + gap_x)
            y = margin_y + row * (card_height + gap_y)
            
            self._create_product_card(slide, product, x, y, card_width, card_height, idx + 1)
        
        print(f"✓ 商品ページ作成: {len(products)}商品")
    
    def _create_product_card(self, slide, product: Dict, x, y, width, height, number: int):
        """個別の商品カードを作成"""
        # 背景枠
        shape = slide.shapes.add_shape(
            1,  # 長方形
            x, y, width, height
        )
        shape.fill.solid()
        shape.fill.fore_color.rgb = RGBColor(255, 255, 255)
        shape.line.color.rgb = RGBColor(*config.COLOR_BORDER)
        shape.line.width = Pt(1)
        
        # 商品番号
        num_box = slide.shapes.add_textbox(x + Inches(0.1), y + Inches(0.1), Inches(0.5), Inches(0.3))
        num_frame = num_box.text_frame
        num_frame.text = f"⑥{number}"
        num_para = num_frame.paragraphs[0]
        num_para.font.size = Pt(12)
        num_para.font.bold = True
        num_para.font.name = config.FONT_NAME
        num_para.font.color.rgb = RGBColor(*config.COLOR_ACCENT)
        
        # 商品名
        name_box = slide.shapes.add_textbox(
            x + Inches(0.1), y + Inches(0.4), width - Inches(0.2), Inches(0.4)
        )
        name_frame = name_box.text_frame
        name_frame.text = product.get('Product_name', '商品名なし')
        name_frame.word_wrap = True
        
        name_para = name_frame.paragraphs[0]
        name_para.font.size = Pt(config.FONT_SIZE_PRODUCT_NAME)
        name_para.font.bold = True
        name_para.font.name = config.FONT_NAME
        
        # 商品画像
        image_file = product.get('画像ファイル名')
        if image_file:
            image_path = config.IMAGES_DIR / image_file
            if image_path.exists():
                try:
                    slide.shapes.add_picture(
                        str(image_path),
                        x + Inches(0.1),
                        y + Inches(0.9),
                        width=Inches(1.8),
                        height=Inches(1.2)
                    )
                except Exception as e:
                    print(f"  ⚠ 画像読み込みエラー: {image_file} - {e}")
        
        # 規格情報
        specs_text = ""
        if product.get('Features3'):  # 産地
            specs_text += f"産地: {product['Features3']}\n"
        if product.get('Features2'):  # 重量・容量
            specs_text += f"容量: {product['Features2']}\n"
        if product.get('Min_quantity'):  # ロット
            specs_text += f"ロット: {product['Min_quantity']}入\n"
        if product.get('温度帯'):
            specs_text += f"温度帯: {product['温度帯']}\n"
        
        if specs_text:
            specs_box = slide.shapes.add_textbox(
                x + Inches(2.1), y + Inches(0.9), width - Inches(2.3), Inches(1.0)
            )
            specs_frame = specs_box.text_frame
            specs_frame.text = specs_text.strip()
            specs_frame.word_wrap = True
            
            for para in specs_frame.paragraphs:
                para.font.size = Pt(9)
                para.font.name = config.FONT_NAME
        
        # 価格
        price = product.get('Price', 0)
        price_box = slide.shapes.add_textbox(
            x + Inches(2.1), y + Inches(1.9), width - Inches(2.3), Inches(0.3)
        )
        price_frame = price_box.text_frame
        price_frame.text = f"¥{price:,.0f} (税別)"
        
        price_para = price_frame.paragraphs[0]
        price_para.font.size = Pt(config.FONT_SIZE_PRICE)
        price_para.font.bold = True
        price_para.font.name = config.FONT_NAME
        price_para.font.color.rgb = RGBColor(*config.COLOR_PRICE)

### 7.4 catalog_generator.py - メイン処理

```python
"""
カタログ生成メインスクリプト
"""
from pathlib import Path
from datetime import datetime
import sys
from typing import List, Dict

import config
from excel_reader import ExcelReader
from pptx_generator import CatalogGenerator

def convert_to_pdf(pptx_path: Path) -> Path:
    """PowerPointファイルをPDFに変換（Windows COM使用）"""
    try:
        import comtypes.client
        
        print("PDF変換中...")
        powerpoint = comtypes.client.CreateObject("Powerpoint.Application")
        powerpoint.Visible = 1
        
        deck = powerpoint.Presentations.Open(str(pptx_path.absolute()))
        pdf_path = pptx_path.with_suffix('.pdf')
        
        # 32 = ppSaveAsPDF
        deck.SaveAs(str(pdf_path.absolute()), 32)
        deck.Close()
        powerpoint.Quit()
        
        print(f"✓ PDF変換完了: {pdf_path}")
        return pdf_path
        
    except Exception as e:
        print(f"⚠ PDF変換エラー: {e}")
        print("  PowerPointファイルのみ生成されました")
        return None

def generate_catalog(
    products: List[Dict],
    makers: List[Dict],
    catalog_config: Dict,
    output_dir: Path = config.OUTPUT_DIR
) -> tuple:
    """カタログを生成"""
    
    # 出力ファイル名
    date_str = datetime.now().strftime("%Y%m%d")
    pptx_filename = config.OUTPUT_FILENAME_FORMAT.format(date=date_str, ext="pptx")
    pptx_path = output_dir / pptx_filename
    
    # カタログ生成開始
    print("\n" + "="*50)
    print("カタログ生成開始")
    print("="*50)
    
    generator = CatalogGenerator()
    
    # 表紙作成
    generator.create_cover_page(catalog_config)
    
    # メーカーごとに商品をグループ化
    makers_dict = {m['メーカーコード']: m for m in makers}
    products_by_maker = {}
    
    for product in products:
        maker_code = product.get('メーカーコード')
        if maker_code not in products_by_maker:
            products_by_maker[maker_code] = []
        products_by_maker[maker_code].append(product)
    
    # メーカー順に並び替え
    sorted_makers = sorted(
        products_by_maker.keys(),
        key=lambda mc: makers_dict.get(mc, {}).get('カタログ表示順', 999)
    )
    
    # 各メーカーのページ生成
    for maker_code in sorted_makers:
        maker = makers_dict.get(maker_code, {})
        maker_products = products_by_maker[maker_code]
        
        # メーカー紹介ページ
        generator.create_maker_intro_page(maker)
        
        # 商品ページ（6商品ずつ）
        for i in range(0, len(maker_products), config.PRODUCTS_PER_PAGE):
            page_products = maker_products[i:i + config.PRODUCTS_PER_PAGE]
            generator.create_product_cards_page(page_products, i // config.PRODUCTS_PER_PAGE + 1)
    
    # PowerPoint保存
    generator.save(pptx_path)
    
    # PDF変換
    pdf_path = convert_to_pdf(pptx_path)
    
    print("\n" + "="*50)
    print("カタログ生成完了！")
    print("="*50)
    print(f"PowerPoint: {pptx_path}")
    if pdf_path:
        print(f"PDF: {pdf_path}")
    print("="*50 + "\n")
    
    return pptx_path, pdf_path

def main():
    """メイン処理"""
    try:
        print("\n" + "="*50)
        print("北海道食材カタログ生成システム")
        print("="*50 + "\n")
        
        # Excelデータ読み込み
        reader = ExcelReader()
        reader.load()
        
        products = reader.get_products()
        makers = reader.get_makers()
        catalog_config = reader.get_config()
        
        # バリデーション
        valid_products, invalid_products = reader.validate_products(products)
        
        if not valid_products:
            print("\n❌ エラー: 有効な商品データがありません")
            sys.exit(1)
        
        # カタログ生成
        pptx_path, pdf_path = generate_catalog(
            valid_products,
            makers,
            catalog_config
        )
        
        print("\n処理が完了しました。何かキーを押して終了してください...")
        input()
        
    except Exception as e:
        print(f"\n❌ エラーが発生しました: {e}")
        import traceback
        traceback.print_exc()
        print("\n何かキーを押して終了してください...")
        input()
        sys.exit(1)

if __name__ == "__main__":
    main()
```

## 付録C: よくある質問（FAQ）

**Q1: 画像が大きすぎてエラーになる**
A: 画像は事前に1920×1080ピクセル以下にリサイズしてください

**Q2: メーカーの順番を変えたい**
A: メーカー情報シートの「カタログ表示順」列の数値を変更

**Q3: 1ページの商品数を変えたい**
A: config.py の PRODUCTS_PER_PAGE を編集

**Q4: 特定の商品だけカタログに載せたくない**
A: 商品情報シートの「カタログ掲載」列を FALSE に設定

**Q5: カタログのデザインを変更したい**
A: pptx_generator.py のフォント・色設定を編集
