## 🚀 Phase 2: 残り67都市の座標計算（実装済み）

### 方法: 逆距離重み付け補間法（IDW）

8都市の正確な座標を基準点として、残り67都市の座標を推定する。

#### アルゴリズム

```python
def calculate_coordinates_idw(target_lat, target_lon, reference_points):
    """
    逆距離重み付け補間法（Inverse Distance Weighting）
    
    近い基準点ほど大きな影響を与える
    重み = 1 / (距離^2)
    """
    
    total_weight = 0
    weighted_x = 0
    weighted_y = 0
    
    for ref in reference_points:
        # 地理的距離を計算（ユークリッド距離の簡易版）
        lat_diff = target_lat - ref['lat']
        lon_diff = target_lon - ref['lon']
        distance = (lat_diff**2 + lon_diff**2) ** 0.5
        
        # 距離が0の場合は直接その座標を返す
        if distance < 0.001:
            return ref['x'], ref['y']
        
        # 重み計算（距離の2乗の逆数）
        weight = 1 / (distance ** 2)
        
        weighted_x += weight * ref['x']
        weighted_y += weight * ref['y']
        total_weight += weight
    
    # 加重平均
    x = int(weighted_x / total_weight)
    y = int(weighted_y / total_weight)
    
    return x, y
```

import pandas as pd

from PIL import Image, ImageDraw, ImageFont

# ===== 設定 =====
BASE_IMAGE = '/mnt/user-data/uploads/hokkaidomap.png'

def romaji_from_kana(kana):
    """カナからローマ字へ簡易変換（完全版は別途実装）"""
    # ここでは簡易的にカナをそのまま返す
    # 実際にはpykakasi等のライブラリを使用
    return kana.title()

def calculate_coordinates_idw(target_lat, target_lon, reference_points):
    """逆距離重み付け補間法で座標を計算"""
    total_weight = 0
    weighted_x = 0
    weighted_y = 0
    
    for ref in reference_points:
        lat_diff = target_lat - ref['lat']
        lon_diff = target_lon - ref['lon']
        distance = (lat_diff**2 + lon_diff**2) ** 0.5
        
        if distance < 0.001:
            return ref['x'], ref['y']
        
        weight = 1 / (distance ** 2)
        weighted_x += weight * ref['x']
        weighted_y += weight * ref['y']
        total_weight += weight
    
    x = int(weighted_x / total_weight)

def draw_star(draw, x, y, size, color='red'):
    """5角星を描画"""
    points = []
    for i in range(10):
        angle = math.pi * 2 * i / 10 - math.pi / 2
        r = size if i % 2 == 0 else size * 0.4
        points.append((x + r * math.cos(angle), y + r * math.sin(angle)))
    draw.polygon(points, fill=color)

def create_location_map(location_data, base_image_path):
    """個別の地名マップ画像を生成"""
    img = Image.open(base_image_path).convert('RGBA')
    
    # 白背景を透明化
    datas = img.getdata()
    newData = []
    for item in datas:
        if item[0] > 240 and item[1] > 240 and item[2] > 240:
            newData.append((255, 255, 255, 0))
        else:
            newData.append(item)
    img.putdata(newData)
    
    draw = ImageDraw.Draw(img)
    
    try:
        font_jp = ImageFont.truetype("/usr/share/fonts/opentype/noto/NotoSansCJK-Bold.ttc", FONT_SIZE_JP)
        font_en = ImageFont.truetype("/usr/share/fonts/opentype/noto/NotoSansCJK-Bold.ttc", FONT_SIZE_EN)
    except:
        font_jp = ImageFont.load_default()
        font_en = ImageFont.load_default()
    
    x = location_data['x']
    y = location_data['y']
    
    draw_star(draw, x, y, STAR_SIZE, color='red')
    
    text_y = y + STAR_SIZE + 20
    
    # 日本語地名
    bbox_jp = draw.textbbox((0, 0), location_data['name'], font=font_jp)
    text_width_jp = bbox_jp[2] - bbox_jp[0]
    text_x_jp = x - text_width_jp/2
    draw.text((text_x_jp, text_y), location_data['name'], 
              fill='white', font=font_jp, stroke_width=4, stroke_fill='black')
    
    # ローマ字
    text_y += FONT_SIZE_JP + 10
    bbox_en = draw.textbbox((0, 0), location_data['romaji'], font=font_en)
    text_width_en = bbox_en[2] - bbox_en[0]
    text_x_en = x - text_width_en/2
    draw.text((text_x_en, text_y), location_data['romaji'], 
              fill='white', font=font_en, stroke_width=3, stroke_fill='black')
    
    return img

def main():
    """メイン処理"""
    
    # 1. 参照点データを読み込み
    with open(REFERENCE_FILE, 'r', encoding='utf-8') as f:
        reference_points = json.load(f)
    
    print(f"参照点: {len(reference_points)}都市")
    
    # 2. CSVから緯度経度データを読み込み
    df = pd.read_csv(CSV_FILE, encoding='shift-jis')
    
    # 3. 全75都市の座標を計算
    all_locations = []
    
    for city_name in TARGET_CITIES:
        # CSVから緯度経度を取得
        match = df[df['name'] == city_name]
        
        if len(match) == 0:
            print(f"⚠️  {city_name}: CSVにデータなし - スキップ")
            continue
        
        lat = match['lat'].iloc[0]
        lon = match['long'].iloc[0]
        kana = match['namekana'].iloc[0]
        
        # 座標を計算（参照点データに含まれる場合はそれを使用）
        ref_match = [r for r in reference_points if r['name'] == city_name]
        if ref_match:
            x = ref_match[0]['x']
            y = ref_match[0]['y']
            source = 'reference'
        else:
            x, y = calculate_coordinates_idw(lat, lon, reference_points)
            source = 'calculated'
        
        # ローマ字を取得
        romaji = ROMAJI_DICT.get(city_name, romaji_from_kana(kana))
        
        location = {
            'name': city_name,
            'romaji': romaji,
            'x': x,
            'y': y,
            'lat': lat,
            'lon': lon,
            'source': source
        }
        
        all_locations.append(location)
    
    print(f"\n座標計算完了: {len(all_locations)}/75都市")
    
    # 4. 座標データを保存
    output_json = os.path.join(OUTPUT_DIR, 'all_75cities_coordinates.json')
    with open(output_json, 'w', encoding='utf-8') as f:
        json.dump(all_locations, f, ensure_ascii=False, indent=2)
    
    print(f"座標データ保存: {output_json}")
    
    # 5. 個別の地名マップ画像を生成
    os.makedirs(OUTPUT_DIR, exist_ok=True)
    
    print("\n=== 地名マップ画像生成中 ===")
    success_count = 0
    
    for location in all_locations:
        try:
            img = create_location_map(location, BASE_IMAGE)
            filename = f"hokkaido_map_{location['romaji'].lower().replace(' ', '_')}.png"
            filepath = os.path.join(OUTPUT_DIR, filename)
            img.save(filepath, 'PNG', optimize=False, compress_level=1)
            
            print(f"✓ {location['name']:12s} ({location['source']:10s}) → {filename}")
            success_count += 1
        except Exception as e:
            print(f"✗ {location['name']:12s} - エラー: {e}")
    
    print(f"\n生成完了: {success_count}/{len(all_locations)}件")
    print(f"出力先: {OUTPUT_DIR}/")

if __name__ == '__main__':
    main()
